CREATE OR REPLACE VIEW monthly_sales_view AS
SELECT 
	'JUNE' AS Month,
    	(total_earningsY-CancelTicket-CancelFood-CancelRental) AS total_earnings,
    	((ticket_total-CancelTicket) / (total_earningsY-CancelTicket-CancelFood-CancelRental)) * 100 AS ticket_percentage,
    	((food_total-CancelFood) / (total_earningsY-CancelTicket-CancelFood-CancelRental)) * 100 AS food_percentage,
    	((rental_total-CancelRental) / (total_earningsY-CancelTicket-CancelFood-CancelRental)) * 100 AS rental_percentage,
    	(rentalnFood_total / (total_earningsY-CancelTicket-CancelFood-CancelRental)) * 100 AS rentalnFood_percentage,
    	(ticketnFood_total / (total_earningsY-CancelTicket-CancelFood-CancelRental)) * 100 AS ticketnFood_percentage
FROM (
    SELECT
        NVL(SUM(grand_total), 0) AS ticket_total
    FROM
        transaction
    WHERE
        EXTRACT(YEAR FROM date_n_time) = '&v_year'
	AND EXTRACT(MONTH FROM date_n_time) = 6
        AND rental_id IS NULL
        AND t_claim_no IS NOT NULL
        AND f_order_no IS NULL
) tt,
(SELECT
        NVL(SUM(grand_total), 0) AS food_total
    FROM
        transaction
    WHERE
        EXTRACT(YEAR FROM date_n_time) = '&v_year'
	AND EXTRACT(MONTH FROM date_n_time) = 6

        AND t_claim_no IS NULL
        AND f_order_no IS NOT NULL
) ft,
(SELECT
        NVL(SUM(grand_total), 0) AS rental_total

(

(SELECT

    FROM
        cancellation c
    JOIN
	ticket_claim_list tcl on c.T_book_id=tcl.T_book_id
    JOIN
	ticket_claim tc on tcl.T_claim_no = tc.T_claim_no
    JOIN
	transaction t on tc.T_claim_no = t.T_claim_no

        EXTRACT(YEAR FROM date_n_time) = '&v_year'
	AND EXTRACT(MONTH FROM date_n_time) = 6
)ct,
(SELECT
        NVL(SUM(refund), 0) AS CancelFood

        cancellation c

	food_order_list fol on c.order_id=fol.order_id
    JOIN
	food_order fo on fol.F_order_no=fo.F_order_no
    JOIN

        EXTRACT(YEAR FROM date_n_time) = '&v_year'

)cf,

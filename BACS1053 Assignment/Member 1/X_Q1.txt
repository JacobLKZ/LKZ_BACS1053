-- Create View
CREATE OR REPLACE VIEW movie_sales_view AS
SELECT 
    m.genre,
    EXTRACT(YEAR FROM tr.date_n_time) AS "Year",
    SUM(CASE WHEN EXTRACT(MONTH FROM tr.date_n_time) = 7 THEN tr.grand_total ELSE 0 END) AS July,
    SUM(CASE WHEN EXTRACT(MONTH FROM tr.date_n_time) = 8 THEN tr.grand_total ELSE 0 END) AS August,
    SUM(tr.grand_total) AS total_sale,
    LAG(SUM(tr.grand_total)) OVER (PARTITION BY m.genre ORDER BY EXTRACT(YEAR FROM tr.date_n_time)) AS prev_year_total,
    SUM(tr.grand_total) - LAG(SUM(tr.grand_total)) OVER (PARTITION BY m.genre ORDER BY EXTRACT(YEAR FROM tr.date_n_time)) AS "Difference"
FROM
    movie m
JOIN 
    movie_screening ms ON m.movie_id = ms.movie_id
JOIN 
    ticket_details td ON ms.show_id = td.show_id
JOIN 
    ticket t ON td.t_details_id = t.t_details_id
JOIN 
    ticket_claim_list tcl ON t.ticket_id = tcl.ticket_id
JOIN 
    ticket_claim tc ON tcl.t_claim_no = tc.t_claim_no
JOIN 
    transaction tr ON tc.t_claim_no = tr.t_claim_no
WHERE 
    EXTRACT(YEAR FROM tr.date_n_time) IN ('&v_year1', '&v_year2','&v_year3') -- Filter for the desired years
GROUP BY 
    m.genre, EXTRACT(YEAR FROM tr.date_n_time)
ORDER BY 
    m.genre, "Year";

--Display view
SELECT * FROM movie_sales_view;
